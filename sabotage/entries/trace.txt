=== the custom method is out of the options while the switch to GET is traced
file: crates/curl-core/src/multi/state.rs
witness: curl-core proto::http::tests::a_custom_method_set_while_a_redirect_is_traced_stands
--- old
    if let Some(said) = said {
        trace::info_line(cx, &said);
    }
--- new
    if let Some(said) = said {
        let custom = cx.easy_mut().options_mut().http.custom_method.take();
        trace::info_line(cx, &said);
        cx.easy_mut().options_mut().http.custom_method = custom;
    }
