pub struct SystemClock(std::time::Instant);

impl SystemClock {
    pub fn new() -> Self {
        SystemClock(std::time::Instant::now())
    }
}

impl Default for SystemClock {
    fn default() -> Self {
        Self::new()
    }
}

impl Clock for SystemClock {
    fn now(&self) -> Instant {
        Instant(micros(self.0.elapsed()))
    }
