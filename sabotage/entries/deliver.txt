# Delivering output to the application's handlers (crates/curl-core/src/
# deliver, handler.rs, the `script:` test protocol), against sendf.c's
# client writers.

=== a handler's replacement of itself is undone after its call
file: crates/curl-core/src/deliver/mod.rs
witness: curl-core deliver::tests::a_handler_replaced_inside_its_own_call_takes_the_next_piece
--- old
            verdict(handler.write(&mut TransferCx::new(cx), piece))?;
--- new
            let mut tcx = TransferCx::new(cx);
            let v = handler.write(&mut tcx, piece);
            let me = tcx.me();
            tcx.options_mut(me).unwrap().callbacks.write = Some(handler);
            verdict(v)?;

=== an included header line reaches the header handler first
file: crates/curl-core/src/deliver/mod.rs
witness: curl-core deliver::tests::included_header_lines_go_to_the_write_handler_first
--- old
        if cx.easy().options().output.include_header {
            self.out.body(cx, line)?;
        }
        self.out.header(cx, line, origin)
--- new
        self.out.header(cx, line, origin)?;
        if cx.easy().options().output.include_header {
            self.out.body(cx, line)?;
        }
        Ok(())
