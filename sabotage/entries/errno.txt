# The operating system's error number a transfer keeps (`os_errno`,
# CURLINFO_OS_ERRNO), against cf-socket.c, easy.c and multi.c.

=== a failed receive keeps no number
file: crates/curl-core/src/conn/layer.rs
witness: curl-sys-abi handle_tests::the_os_errno_is_the_last_socket_failure_s

                if let Err(e) = &got {
                    cx.socket_failed(e);

--- new
