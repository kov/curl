# Transfers over a Unix socket (CURLOPT_UNIX_SOCKET_PATH,
# CURLOPT_ABSTRACT_UNIX_SOCKET, a SOCKS proxy at localhost with a path).

=== a Unix socket's path is compared in any case
file: crates/curl-core/src/conn/mod.rs
witness: curl-core conn::tests::unix_sockets_are_told_apart_as_c_tells_them
--- old
                PeerAt::Unix | PeerAt::AbstractUnix => self.host == other.host,
--- new
                PeerAt::Unix | PeerAt::AbstractUnix => self.host.eq_ignore_ascii_case(&other.host),

=== a Unix socket is the same as a host of its path
file: crates/curl-core/src/conn/mod.rs
witness: curl-core conn::tests::unix_sockets_are_told_apart_as_c_tells_them
--- old
        *scheme == other.scheme
            && *port == other.port
            && *at == other.at
--- new
        *scheme == other.scheme
            && *port == other.port
            && (*at == other.at || *at != PeerAt::AbstractUnix)
