# The fallible reference-counted pointers (crates/curl-alloc/src/rc.rs),
# which every handler is: one block, its count and the value, a trait
# object among them.

=== a refused block forgets the value it was to hold
file: crates/curl-alloc/src/rc.rs
witness: curl-alloc rc::tests::a_refused_block_drops_its_value
features: debug-hooks
--- old
        let block = try_box(SharedBlock {
            count: C::one(),
            value,
        })?;
--- new
        let block = crate::try_box_or_back(SharedBlock {
            count: C::one(),
            value,
        })
        .map_err(|b| {
            std::mem::forget(b);
            OutOfMemory
        })?;
