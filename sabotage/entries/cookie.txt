=== a cookie is stored after its line is traced
file: crates/curl-core/src/proto/http.rs
witness: curl-core proto::http::tests::a_cookie_set_goes_with_the_next_request
--- old
        // A cookie is stored before its line is traced (`http_header`).
        if line.kind == LineKind::Header
            && let Some(value) = strip_name(line.bytes, b"Set-Cookie:")
        {
            let site = cookie_site(&self.cookie_host, &self.target, &self.peer);
            cookies::set_by_server(cx, value, &site)?;
        }
        let early = line.early.map_err(|e| head_failed(cx, &e));
        // What reading its transfer codings did, traced before the line, as
        // C reads them as it handles it (`http_header_t`).
        if let Some(tale) = line.codings {
            trace_codings(cx, tale);
        }
        trace::bytes(cx, DebugKind::HeaderIn, line.bytes);
--- new
        let early = line.early.map_err(|e| head_failed(cx, &e));
        if let Some(tale) = line.codings {
            trace_codings(cx, tale);
        }
        trace::bytes(cx, DebugKind::HeaderIn, line.bytes);
        if line.kind == LineKind::Header
            && let Some(value) = strip_name(line.bytes, b"Set-Cookie:")
        {
            let site = cookie_site(&self.cookie_host, &self.target, &self.peer);
            cookies::set_by_server(cx, value, &site)?;
        }

=== the cookie jar is renamed into place by names std copies
file: crates/curl-core/src/save.rs
witness: curl-sys-abi handle_tests::flushes_a_jar_named_long_without_copying_the_name
features: debug-hooks
--- old
    let renamed = written.and_then(|()| Ok(fs::rename(&temp, &target)?));
--- new
    let renamed = written.and_then(|()| {
        use std::os::unix::ffi::OsStrExt as _;
        let (from, to) = (temp.to_bytes(), target.to_bytes());
        Ok(std::fs::rename(std::ffi::OsStr::from_bytes(from), std::ffi::OsStr::from_bytes(to))?)
    });
